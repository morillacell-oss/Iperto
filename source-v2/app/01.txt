async function loadOffers(){
  $('#offersStatus').style.display='block';
  const [{data:offers,error},{data:businesses},{data:stats}]=await Promise.all([
    supabase.from('offers').select('*').in('status',['scheduled','open','closed','sold_out']).order('featured',{ascending:false}).order('opens_at',{ascending:true}),
    supabase.from('businesses').select('*').eq('status','approved'),
    supabase.from('offer_stats').select('*')
  ]);
  if(error){console.error(error);$('#offersStatus').innerHTML='Não conseguimos carregar as ofertas agora.';return}
  state.offers=offers||[];(businesses||[]).forEach(b=>state.businesses.set(b.id,b));(stats||[]).forEach(s=>state.stats.set(s.offer_id,s));
  $('#offersStatus').style.display='none';
  const total=state.offers.reduce((sum,o)=>sum+Number(state.stats.get(o.id)?.waitlist_count||o.demo_seed_count||0),0);$('#heroQueue').textContent=total||175;
  renderOffers();
}
